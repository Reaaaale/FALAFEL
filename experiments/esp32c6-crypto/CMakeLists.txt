cmake_minimum_required(VERSION 3.16)

option(FALAFEL_SEC003_HOST_TESTS "Native rehearsal of the SAME experiment suite" OFF)
if(FALAFEL_SEC003_HOST_TESTS)
    project(falafel_sec003_host C)
    set(CMAKE_C_STANDARD 11)
    set(CMAKE_C_STANDARD_REQUIRED ON)
    set(FALAFEL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
    set(SEC003_MBEDTLS "${FALAFEL_ROOT}/.cache/sec002/esp-idf-v5.5.6/components/mbedtls/mbedtls")
    set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
    set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory("${SEC003_MBEDTLS}" mbedtls)
    foreach(kind raw probed)
        add_library(sec003_provider_${kind} STATIC "${FALAFEL_ROOT}/shared/crypto/falafel_crypto_mbedtls.c")
        target_include_directories(sec003_provider_${kind} PUBLIC "${FALAFEL_ROOT}/shared/include")
        target_link_libraries(sec003_provider_${kind} PUBLIC MbedTLS::mbedcrypto)
    endforeach()
    target_compile_definitions(sec003_provider_probed PRIVATE mbedtls_entropy_func=sec003_entropy_func)
    add_executable(sec003_host main/host_main.c main/sec003_suite.c main/sec003_probe.c)
    target_include_directories(sec003_host PRIVATE main)
    target_link_libraries(sec003_host PRIVATE sec003_provider_probed)
    add_executable(sec003_provider_regression "${FALAFEL_ROOT}/test/test_crypto.c")
    target_link_libraries(sec003_provider_regression PRIVATE sec003_provider_raw)
    foreach(target sec003_host sec003_provider_regression sec003_provider_raw sec003_provider_probed)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endforeach()
    enable_testing()
    add_test(NAME sec003_native_rehearsal COMMAND sec003_host)
    add_test(NAME sec003_production_provider_regression COMMAND sec003_provider_regression)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME sec003_log_analysis COMMAND ${Python3_EXECUTABLE} "${FALAFEL_ROOT}/test/test_sec003_logs.py")
else()
    include($ENV{IDF_PATH}/tools/cmake/project.cmake)
    project(falafel_esp32c6_crypto)
endif()
